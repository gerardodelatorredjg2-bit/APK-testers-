===== MobileLightOptimizer.gd =====
extends Node


@export var max_lights: int = 8


@export var update_interval: float = 0.1

var _timer: float = 0.0

func _process(delta: float) -> void :
	_timer += delta
	if _timer >= update_interval:
		_timer = 0.0
		_optimize_scene_lights()

func _optimize_scene_lights() -> void :

	var camera: = get_viewport().get_camera_3d()
	if not camera:
		return


	var lights: Array[Light3D] = []
	_get_lights_recursive(get_tree().root, lights)


	if lights.size() <= max_lights:
		for l in lights:
			l.visible = true
		return


	var cam_pos: = camera.global_position
	lights.sort_custom( func(a: Light3D, b: Light3D) -> bool:
		return a.global_position.distance_squared_to(cam_pos) < b.global_position.distance_squared_to(cam_pos)
	)


	for i in range(lights.size()):
		lights[i].visible = (i < max_lights)

func _get_lights_recursive(node: Node, lights_list: Array[Light3D]) -> void :

	if node is Light3D and not node is DirectionalLight3D:
		lights_list.append(node)

	for child in node.get_children():
		_get_lights_recursive(child, lights_list)

===== AllInOneMobileOptimizer.gd =====
extends Node


@export var max_lights: int = 8


@export var update_interval: float = 0.15


@export var target_mobile_fps: int = 60

var _timer: float = 0.0

func _ready() -> void :



	if OS.has_feature("mobile") or OS.get_name() in ["Android", "iOS"]:

		Engine.max_fps = target_mobile_fps


		RenderingServer.positional_soft_shadow_filter_set_quality(RenderingServer.SHADOW_QUALITY_HARD)
		RenderingServer.directional_soft_shadow_filter_set_quality(RenderingServer.SHADOW_QUALITY_HARD)


		_strip_heavy_environment_effects()

func _process(delta: float) -> void :
	_timer += delta
	if _timer >= update_interval:
		_timer = 0.0
		_update_nearest_lights()

func _update_nearest_lights() -> void :
	var camera: = get_viewport().get_camera_3d()
	if not camera:
		return


	var lights: Array[Light3D] = []
	_find_lights_recursive(get_tree().root, lights)


	if lights.size() <= max_lights:
		for light in lights:
			light.visible = true
		return

	var cam_pos: = camera.global_position

	var cam_forward: = - camera.global_transform.basis.z.normalized()

	lights.sort_custom( func(a: Light3D, b: Light3D) -> bool:
		var dir_a: = (a.global_position - cam_pos)
		var dir_b: = (b.global_position - cam_pos)

		var dist_a: = dir_a.length_squared()
		var dist_b: = dir_b.length_squared()


		var in_front_a: = cam_forward.dot(dir_a.normalized()) > -0.2
		var in_front_b: = cam_forward.dot(dir_b.normalized()) > -0.2


		if in_front_a != in_front_b:
			return in_front_a


		return dist_a < dist_b
	)


	for i in range(lights.size()):
		lights[i].visible = (i < max_lights)

func _find_lights_recursive(node: Node, light_array: Array[Light3D]) -> void :

	if node is Light3D and not node is DirectionalLight3D:
		light_array.append(node)

	for child in node.get_children():
		_find_lights_recursive(child, light_array)

func _strip_heavy_environment_effects() -> void :
	var world_env: = _find_world_environment(get_tree().root)
	if world_env and world_env.environment:
		var env: = world_env.environment
		env.ssao_enabled = false
		env.ssr_enabled = false
		env.glow_enabled = false
		env.sdfgi_enabled = false

func _find_world_environment(node: Node) -> WorldEnvironment:
	if node is WorldEnvironment:
		return node
	for child in node.get_children():
		var found: = _find_world_environment(child)
		if found:
			return found
	return null

===== controls_android.gd =====
extends Node2D

var controls: Dictionary = {}

func _ready() -> void :

	_collect_controls()

	if Globals.touch_controls_defaults.is_empty():
		Globals.touch_controls_defaults = capture_layout()

	apply_layout(Globals.settings.get("touch_controls", {}))

func _collect_controls() -> void :

	controls.clear()

	for c in $CanvasLayer.get_children():
		if c is TouchScreenButton and c.texture_normal != null:
			controls[c.name] = c
		elif c is VirtualJoystick:
			controls[c.name] = c

func _input(event: InputEvent) -> void :





	if event is InputEventMouseButton and event.pressed:
		if _is_point_inside_any_button(event.position):
			get_viewport().set_input_as_handled()

func _is_point_inside_any_button(point: Vector2) -> bool:
	for c in controls.values():
		if c is TouchScreenButton and c.is_visible_in_tree():
			if _is_point_in_rect(point, c.global_position, c.texture_normal.get_size() * c.scale):
				return true
	return false

func _is_point_in_rect(point: Vector2, origin: Vector2, size: Vector2) -> bool:
	return point.x >= origin.x and point.x <= origin.x + size.x and point.y >= origin.y and point.y <= origin.y + size.y

func get_controls() -> Dictionary:
	return controls

func set_editing(editing: bool) -> void :

	for c in controls.values():
		if editing:
			c.process_mode = Node.PROCESS_MODE_DISABLED
		else:
			c.process_mode = Node.PROCESS_MODE_INHERIT

func capture_layout() -> Dictionary:

	var layout: = {}

	for name in controls:
		var c = controls[name]
		layout[name] = {
			"pos": [c.global_position.x, c.global_position.y], 
			"scale": [c.scale.x, c.scale.y], 
			"opacity": c.modulate.a
		}

	return layout

func apply_layout(layout: Dictionary) -> void :

	for name in layout:
		if controls.has(name):
			var c = controls[name]
			var data = layout[name]
			c.global_position = Vector2(data["pos"][0], data["pos"][1])
			c.scale = Vector2(data["scale"][0], data["scale"][1])
			c.modulate.a = data.get("opacity", 1.0)

func apply_default_layout() -> void :

	apply_layout(Globals.touch_controls_defaults)

func save_layout() -> void :

	Globals.settings["touch_controls"] = capture_layout()
	Globals.save_current_settings()
